Integrating Eyes, Teeth & Tongue into MHR (Three.js / WebGPU)

Copy-paste pattern — now backed by mhr-eyes-oral-module.js in this folder

1. Parent to MHR Bones
Attach eyeLeftBone, eyeRightBone, lids + upperJawGroup to MHR's head bone. Attach lowerJawGroup to MHR's jaw bone.
2. Holistic Dental Morphs
Instead of 28 individual tooth bones, crookedness, diastema gap, and overbite are encoded as whole-arch vertex morph targets — 0 extra bones, 4 floats.
3. Occlusion Gating
Skip updating oral morph targets or tongue compute passes whenever jawOpen < 0.02 and betaF[mouthOpen] < 0.05. Groups are hidden via visible=false. 
import { createMHREyesAndMouthModule } from './mhr-eyes-oral-module.js';

// 1. Instantiate the modular attachment system and bind to your MHR skeleton
const headBone = mhrSkinnedMesh.skeleton.getBoneByName('head');
const jawBone  = mhrSkinnedMesh.skeleton.getBoneByName('jaw');

const oralEyeSystem = createMHREyesAndMouthModule();
headBone.add(
  oralEyeSystem.eyeLeftBone, oralEyeSystem.eyeRightBone,
  oralEyeSystem.lidLeft.group, oralEyeSystem.lidRight.group,
  oralEyeSystem.upperJawGroup
);
jawBone.add(oralEyeSystem.lowerJawGroup); // lower teeth + tongue + mouth floor ride on the mandible!

// 2. Per-frame sync with MHR's 204 parameters & 72 FACS expression blendshapes
function updateCharacterSubRigs(gazePitch, gazeYaw, mhrExpressionBeta72) {
  const { upperLidFollow, lowerLidFollow } = oralEyeSystem.setGaze(gazeYaw, gazePitch, 0);
  mhrExpressionBeta72[FACS_IDX.EYE_LOOK_UP]   = Math.max(0, upperLidFollow);
  mhrExpressionBeta72[FACS_IDX.EYE_LOOK_DOWN] = Math.max(0, lowerLidFollow);

  // Occlusion culling: hide/skip oral cavity work when mouth is closed
  const mouthOpenBeta = mhrExpressionBeta72[FACS_IDX.JAW_OPEN] ?? 0;
  const jawDeg = THREE.MathUtils.radToDeg(jawBone.rotation.x);
  const isVisible = oralEyeSystem.updateOcclusion(jawDeg, mouthOpenBeta);
  if (isVisible) {
    oralEyeSystem.setTeeth({ count: 14, crooked: 0, gap: 0, overbite: 0, fangs: 0 });
    oralEyeSystem.setTongue({ out: 0.2, curl: 0, swing: 0, width: 0, roll: 0, thick: 0 });
  }
}
